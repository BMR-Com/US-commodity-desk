            {
                nStr += '';
                x = nStr.split('.');
                x1 = x[0];
                x2 = x.length > 1 ? '.' + x[1] : '';
                var rgx = /(\d+)(\d{3})/;

                var n1, n2;
                num = num + '' || '';
                // works for integer and floating as well
                n1 = num.split('.');
                n2 = n1[1] || null;
                n1 = n1[0].replace(/(\d)(?=(\d\d)+\d$)/g, "$1,");
                num = n2 ? n1 + '.' + n2 : n1;
                return num;

    $("#spot_rate_card_date").flatpickr({
        dateFormat: 'd M Y',
        defaultDate: "29 Oct 2026",
        disableMobile: "true",
        onChange: function (selectedDates, dateStr) {
            loadSpotRateCard(dateStr);
        }
    });

    function loadSpotRateCard(date) {
        $.ajax({
            url: site_url + "details/cai/spot-rate-card/by/date",
            type: 'POST',
            data: { date: date },
            dataType: 'json',
            success: function (data) {
                $("#spot_rate_date_text").text(data.date);
                var rows = '';
                if (data.list.length === 0) {
                    rows = '<tr><td colspan="5" class="text-center">No data available</td></tr>';
                } else {
                    data.list.forEach(function (row) {
                        var badgeClass, label;
                        if (row.price == 0) { badgeClass = 'zero'; label = '0'; }
                        else if (row.diff > 0) { badgeClass = 'up'; label = '▲ ' + Math.abs(row.diff); }
                        else if (row.diff < 0) { badgeClass = 'down'; label = '▼ ' + Math.abs(row.diff); }
                        else { badgeClass = 'flat'; label = 'unch'; }

                        rows += '<tr><td><b>' + row.grade_standard + '</b></td><td>' + row.staple + '</td><td>' +
                                row.growth + '</td><td>' + row.price.toLocaleString() + '</td><td>' +
                                '<span class="badge ' + badgeClass + '">' + label + '</span></td></tr>';
                    });
                }
                $("#spot_rate_tbody").html(rows);
            }
        });
    }
</script>
