                    macBalesArrived = Math.round(parseFloat(cummulative[i].bales_arrived)) + Math.round(parseFloat(adjustment[i].bales_arrived));
                    macBalesArrivedTones = Math.round(parseFloat(cummulative[i].bales_matric_tones)) + Math.round(parseFloat(adjustment[i].bales_matric_tones));
                } else {
                    cbalesArrived += Math.round(parseFloat(cummulative[i].bales_arrived));
                    cbalesArrivedTones += Math.round(parseFloat(cummulative[i].bales_matric_tones));
                    macBalesArrived = Math.round(parseFloat(cummulative[i].bales_arrived));
                    macBalesArrivedTones = Math.round(parseFloat(cummulative[i].bales_matric_tones));
                }

                table += `<tr >
                            <td colspan="1" style="text-align: center;">${(i + 1)}</td>
                            <td colspan="2" style="text-align: left;">${listOf[i].name}</td>
                            <td colspan="2" style="text-align: right;">${formatValue(listOf[i].bales_arrived)}</td>
                            <td colspan="2" style="text-align: right;">${formatValue(listOf[i].bales_matric_tones)}</td>
                            <td colspan="2" style="text-align: right;">${formatValue(macBalesArrived)}</td>
                            <td colspan="2" style="text-align: right;">${formatValue(macBalesArrivedTones)}</td>
                        </tr>`;
            }

            var totals = `<tr >
                            <td colspan="1" style="text-align: center;"></td>
                            <td colspan="2" style="text-align: left;"><strong>TOTAL</strong></td>
                            <td colspan="2" style="text-align: right;"><strong>${formatValue(dbalesArrived)}</strong></td>
                            <td colspan="2" style="text-align: right;"><strong>${formatValue(dbalesArrivedTones)}</strong></td>
                            <td colspan="2" style="text-align: right;"><strong>${formatValue(cbalesArrived)}</strong></td>
                            <td colspan="2" style="text-align: right;"><strong>${formatValue(cbalesArrivedTones)}</strong></td>
                        </tr>`;
            $(".spot_rate").html(table + totals + seasonsDisplay);

        });
    }

    function formatValue(value, u) {
        if (u == 0) {
            //170kg
            var uVal = '5.882353';
            var d = parseFloat(value) * parseFloat(uVal);
            if (d == 0) {
                return "";
            } else {
                return addFormat(d.toFixed(2), 0);
            }
        } else if (u == 1) {
            //480 lbs
            var uVal = '4.592917';
            var d = parseFloat(value) * parseFloat(uVal);
            if (d == 0) {
                return "";
            } else {
                return addFormat(d.toFixed(3), 0);
            }
        } else if (u == 2) {
            //Tones
            var d = parseFloat(value);
            if (d == 0) {
                return "";
            } else {
                return addFormat(d.toFixed(2), 2);
            }
        } else {
            //no unit
            var d = parseFloat(value);
            if (d == 0) {
                return "";
            } else {
                return addFormat(d.toFixed(0), 0);
            }
        }
    }
