        $("tr#last_tr").nextAll().remove();
        
        var weight_type = $("#weight_type>button.active").attr('id');
        var fixed_decimal = 0;

            fixed_decimal = 0;
        }else if(weight_type == 'lbs'){
            fixed_decimal = 3;
        }else{
            fixed_decimal = 2;
        }
        var year = $(elem).val();
        var form = new FormData();
        form.append("year", year);
        form.append("weight_type", weight_type);
        $.ajax({
            type: "POST",
            url: site_url + "news_reports/cab_area_production/filter",
            data: form,
            processData: false,
            dataType: 'json',
            contentType: false,
            cache: false,
            timeout: 600000,
            success: function (data) {
                var sethtml = ``;
                if(data.status == 1){
                    $.each(data.rec, function (i, obj) {
                        sethtml += `<tr class="thead-title">
                                        <th colspan="6" style="text-align: left;"><h4><strong>${i.toUpperCase() == "OTHER" ? "OTHERS":i.toUpperCase()+" ZONE"}</strong></h4></th>
                                    </tr>`;
                        $.each(obj, function (i1, obj1) {
                            if(obj1.state == 'TOTAL' ){
                                if(fixed_decimal > 0){
                                    sethtml += `<tr class="advisory-total">
                                                    <td><strong>${obj1.state}</strong></td>
                                                    <td style="text-align: right;"><strong>${addFormat(obj1.area, fixed_decimal)}</strong></td>`;
                                }else{

                                }
                                
                                sethtml += `<td style="text-align: right;"><strong>${addFormat(obj1.pressed_bales, fixed_decimal)}</strong></td>
                                                <td style="text-align: right;"><strong>${addFormat(obj1.loose_cotton, fixed_decimal)}</strong></td>
                                                <td style="text-align: right;"><strong>${addFormat(obj1.total, fixed_decimal)}</strong></td><td style="text-align: right;"><strong>${addFormat(obj1.yield, 2)}</strong></td>`;
                                sethtml += `</tr>`;
                            }else{
                                sethtml += `<tr><td>${obj1.state}</td>`;
                                if(fixed_decimal > 0){
                                    sethtml += `<td style="text-align: right;">${addFormat(obj1.area, fixed_decimal)}</td>`;
                                }else{
                                    sethtml += `<td style="text-align: right;">${addFormat(obj1.area, 2)}</td>`;
                                }
                                sethtml += `<td style="text-align: right;">${addFormat(obj1.pressed_bales, fixed_decimal)}</td>

                                            <td style="text-align: right;">${addFormat(obj1.total, fixed_decimal)}</td>`;

                                }else{
                                    sethtml += `<td style="text-align: right;">${addFormat(obj1.yield, 2)}</td>`;
                                }
                            }
                        });
                    });
                    sethtml += `<tr></tr><tr class="advisory-total"><td><strong>GRAND TOTAL</strong></td>`;
                    $.each(data.grand, function (i, obj) {
                        if(fixed_decimal > 0){
                            sethtml += `<td style="text-align: right;"><strong>${addFormat(obj, 2)}</strong></td>`;   
                        }else{
                            if(i=='pressed_bales' || i=='loose_cotton' || i=='total'){
                                sethtml += `<td style="text-align: right;"><strong>${addFormat(obj, fixed_decimal)}</strong></td>`;    
                            }else if(i=='area' || i=='yield' ){
                                sethtml += `<td style="text-align: right;"><strong>${addFormat(obj, 2)}</strong></td>`;
                            }else{
                                sethtml += `<td style="text-align: right;"></td>`;
                            }
                        }
                    });
                    
                    $("#remarks_msg").html(data.remarks_msg);
                }else{
                   alert(data.message);
                }
                $(".sethtml").append(sethtml);
            },
            error: function () {
                console.log();
            }
        });
