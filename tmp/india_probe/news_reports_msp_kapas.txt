                if(data.status == 1){

                        $("#remark").html(" <b>Remark :-</b> "+data.remark);    
                    }else{
                        $("#remark").html("");

                        var a = obj.a;

                            var b = (obj.b) ;
                            $(".class_hide").show();

                            $(".class_hide").hide();
                            $("#extra_column").attr("colspan","4");

                        $("#"+i+"_a").html(numberFormat(a));

                    });
